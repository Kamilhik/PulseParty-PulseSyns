cmake_minimum_required(VERSION 3.24)
project(pulseparty_discord LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DISCORD_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Extracted official Discord Social SDK")
find_path(DISCORD_INCLUDE discordpp.h PATHS "${DISCORD_SDK_ROOT}/include" "${DISCORD_SDK_ROOT}" NO_DEFAULT_PATH REQUIRED)
find_library(DISCORD_LIBRARY NAMES discord_partner_sdk PATHS "${DISCORD_SDK_ROOT}/lib/release" "${DISCORD_SDK_ROOT}/lib" "${DISCORD_SDK_ROOT}/lib/x86_64" "${DISCORD_SDK_ROOT}/lib/x86_64/release" NO_DEFAULT_PATH REQUIRED)
add_executable(pulseparty-discord main.cpp)
target_include_directories(pulseparty-discord PRIVATE "${DISCORD_INCLUDE}")
target_link_libraries(pulseparty-discord PRIVATE "${DISCORD_LIBRARY}")
if(MSVC)
  target_compile_options(pulseparty-discord PRIVATE /utf-8)
endif()
if(WIN32)
  file(GLOB_RECURSE DISCORD_RUNTIME "${DISCORD_SDK_ROOT}/*discord_partner_sdk.dll")
  list(LENGTH DISCORD_RUNTIME DLL_COUNT)
  if(NOT DLL_COUNT EQUAL 1)
    message(FATAL_ERROR "Keep exactly one matching Windows x64 SDK runtime in DISCORD_SDK_ROOT")
  endif()
  add_custom_command(TARGET pulseparty-discord POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DISCORD_RUNTIME}" "$<TARGET_FILE_DIR:pulseparty-discord>")
else()
  set_target_properties(pulseparty-discord PROPERTIES BUILD_RPATH "$ORIGIN;@loader_path")
endif()
